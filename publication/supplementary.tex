\section*{S1. Parser source}

\subsection*{S1.1 V1 parser (single-pass)}
\begin{lstlisting}
TOOL_CALL_PATTERN = re.compile(
    r"<tool_call>\s*(\{.*?\})\s*</tool_call>",
    re.DOTALL,
)
# On malformed input: no match; server returns raw text as final answer.
\end{lstlisting}

\subsection*{S1.2 V2 parser (two-pass, used in E4 and V2)}
\begin{lstlisting}
TOOL_CALL_CLOSED = re.compile(
    r"<tool_call>\s*(\{.*?\})\s*</tool_call>",
    re.DOTALL,
)
TOOL_CALL_OPEN = re.compile(
    r"<tool_call>\s*(\{.*?\})",
    re.DOTALL,
)
# Pass 1: strict. Pass 2 (fallback): first JSON after any <tool_call>.
# Known limitation: Pass 2 recovers only the first call in a malformed
# multi-call output and may truncate nested arguments.
\end{lstlisting}
